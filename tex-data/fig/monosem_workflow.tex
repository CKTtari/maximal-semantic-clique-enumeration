\node[rule, fill=paperblue!8] (parent) at (0,0)
  {\textbf{Feasible parent.} For feasible $D$, delete the vertex $d(D)$
   with minimum total similarity. Then
   $\avgsim(D\setminus\{d(D)\})\geq\avgsim(D)$.};
\node[rule, fill=papergold!11] (owner) at (3.65,0)
  {\textbf{Unique ownership.} Traverse $C\rightarrow D=C\cup\{x\}$ only
   if $x=d(D)$. Thus each state has one parent and needs no visited table.};

\node[phase] at (-1.67,-1.02) {Owned path to $C_1$};
\node[state, fill=paperblue!7] (e24) at (-0.70,-1.72)
  {$\{v_2,v_4\}$\\$1.000$};

  {$\{v_1,v_2,v_4\}$\\$0.867$};

  {$C_1$\\$0.693$};
\draw[accepted] (e24) -- node[above,font=\tiny,align=center] {$+v_1$\\owned} (t124);
\draw[accepted] (t124) -- node[above,font=\tiny,align=center] {$+v_3$\\owned} (c1);

\node[phase] at (-1.67,-2.53) {Two paths to $C_2$};
\node[route, fill=papergreen!10] (keep) at (0,-3.22)
  {\textbf{Traverse}\\
   $\{v_5,v_6\}+v_3\rightarrow C_2$\\
   $v_3=d(C_2)$};
\node[route, fill=paperred!7] (drop) at (3.65,-3.22)
  {\textbf{Stop duplicate path}\\
   $\{v_3,v_5\}+v_6\rightarrow C_2$\\
   $v_6\neq d(C_2)=v_3$};

\node[route, fill=paperblue!7, text width=68.5mm, minimum height=8mm]
  (result) at (1.825,-4.30)
