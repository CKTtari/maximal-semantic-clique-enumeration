\documentclass[tikz,border=3pt]{standalone}
\usepackage{newtxtext,newtxmath}
\usetikzlibrary{arrows.meta,positioning}
\definecolor{paperink}{HTML}{3F4852}
\definecolor{papergray}{HTML}{EEF1F4}
\definecolor{paperamber}{HTML}{A66A14}
\definecolor{papergreen}{HTML}{287A4D}
\begin{document}
\begin{tikzpicture}[
  font=\scriptsize,
  >=Latex,
  box/.style={draw=paperink, rounded corners=1.5pt, line width=0.55pt,
    align=center, minimum height=9mm, inner sep=1.4mm},
  wide/.style={box, text width=66mm},
  branch/.style={box, text width=29mm, minimum height=12mm},
  accepted/.style={fill=papergreen!11, draw=papergreen},
  bounded/.style={fill=paperamber!9, draw=paperamber},
  structural/.style={fill=papergray},
  arrow/.style={->, line width=0.9pt, draw=paperink}
]
\node[wide, structural] (bk) at (1.85,0)
  {\textbf{Structural MCE}\quad
   $M_1=\{v_1,v_2,v_3,v_4\}$,\quad
   $M_2=\{v_3,v_4,v_5,v_6\}$};

\node[branch, accepted] (m1) at (0,-1.45)
  {\textbf{$M_1$ is feasible}\\
   $\operatorname{avgSim}(M_1)=0.693$\\
   accept $C_1=M_1$};
\node[branch, bounded] (m2) at (3.70,-1.45)
  {\textbf{$M_2$ is infeasible}\\
   $\operatorname{avgSim}(M_2)=0.427$\\
   bounded subset search};
\draw[arrow] ([xshift=-18.5mm]bk.south) -- (m1.north);
\draw[arrow] ([xshift=18.5mm]bk.south) -- (m2.north);

\node[branch, accepted] (c2) at (3.70,-3.05)
  {\textbf{Feasible subset}\\
   $C_2=\{v_3,v_5,v_6\}$\\
   $\operatorname{avgSim}(C_2)=0.693$};
\draw[arrow] (m2) -- (c2);

\node[wide, structural] (filter) at (1.85,-4.50)
  {\textbf{Global filter}\quad deduplicate candidates and remove strict subsets};
\draw[arrow] (m1.south) -- (m1.south |- filter.north);
\draw[arrow] (c2.south) -- (c2.south |- filter.north);

\node[box, accepted, text width=38mm, minimum height=8mm] (out) at (1.85,-5.55)
  {\textbf{Output}\quad $\mathcal R=\{C_1,C_2\}$};
\draw[arrow] (filter) -- (out);
\end{tikzpicture}
